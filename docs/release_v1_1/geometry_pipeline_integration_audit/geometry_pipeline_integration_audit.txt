   295	def validate_stage_contract_registry() -> None:
   296	    """
   297	    Fail early if pipeline stages and contracts diverge.
   298	    """
   299	
   300	    stage_names = [
   301	        x.name
   302	        for x in STAGES
   303	    ]
   304	
   305	    if len(
   306	        stage_names
   307	    ) != len(
   308	        set(
   309	            stage_names
   310	        )
   311	    ):
   312	        raise RuntimeError(
   313	            "Duplicate production stage names."
   314	        )
   315	
   316	    contract_names = set(
   317	        STAGE_CONTRACTS
   318	    )
   319	
   320	    expected_names = set(
   321	        stage_names
   322	    )
   323	
   324	    missing = sorted(
   325	        expected_names
   326	        -
   327	        contract_names
   328	    )
   329	
   330	    extra = sorted(
   331	        contract_names
   332	        -
   333	        expected_names
   334	    )
   335	
   336	    if missing or extra:
   337	        raise RuntimeError(
   338	            "Stage contract registry mismatch: "
   339	            f"missing={missing}, extra={extra}"
   340	        )
   341	
   342	    for name in stage_names:
   343	
   344	        contract = (
   345	            STAGE_CONTRACTS[
   346	                name
   347	            ]
   348	        )
   349	
   350	        if contract.name != name:
   351	            raise RuntimeError(
   352	                "Stage contract name mismatch: "
   353	                f"{name} != {contract.name}"
   354	            )

   357	def _sha256_file(path: str | Path) -> str:
   358	    """Return the SHA256 digest of a file."""
   359	
   360	    p = Path(path)
   361	
   362	    h = hashlib.sha256()
   363	
   364	    with p.open("rb") as f:
   365	        while True:
   366	            block = f.read(1024 * 1024)
   367	
   368	            if not block:
   369	                break
   370	
   371	            h.update(block)
   372	
   373	    return h.hexdigest()
