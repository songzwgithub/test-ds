    90	def cmd_plan(args):
    91	    cfg, path = load_config(args.config)
    92	    if args.ndate is not None:
    93	        ndate = int(args.ndate)
    94	    else:
    95	        raw = cfg_get(cfg, "runtime.planning_ndate", None)
    96	        if raw not in (None, "", "auto"):
    97	            ndate = int(raw)
    98	        else:
    99	            _, stack = _open_stack(cfg, path)
   100	            ndate = len(stack.dates)
   101	    plan = build_runtime_plan(
   102	        ndate=ndate,
   103	        memory_fraction=float(cfg_get(cfg, "runtime.memory_fraction", 0.85)),
   104	        requested_cpu=_requested_cpu(cfg),
   105	        max_solver_size=_runtime_solver_size(cfg, ndate),
   106	    )
   107	    payload = plan.as_dict()
   108	    payload["ndate"] = int(ndate)
   109	    print(json.dumps(payload, indent=2))

   112	def _validate_project_choices(cfg, stack):
   113	    errors = []
   114	    geometric_ref = cfg_get(cfg, "phase_correction.geometric_reference_date", None)
   115	    if geometric_ref in (None, ""):
   116	        errors.append("phase_correction.geometric_reference_date is not set")
   117	    elif str(geometric_ref).lower() != "auto" and str(geometric_ref) not in stack.dates:
   118	        errors.append("phase_correction.geometric_reference_date is not present in RSLC_tab")
   119	    if str(cfg_get(cfg, "reference.method", "radar_window")) == "radar_window":
   120	        row = cfg_get(cfg, "reference.radar_window.center_row", None)
   121	        col = cfg_get(cfg, "reference.radar_window.center_col", None)
   122	        if row is None or col is None:
   123	            errors.append("reference.radar_window.center_row/center_col must be set")
   124	    if errors:
   125	        raise RuntimeError("Project-specific configuration is incomplete:\n  - " + "\n  - ".join(errors))

   193	def build_parser():
   194	    p = argparse.ArgumentParser(prog="pypsds", description="pyPSDS-GAMMA production CLI")
   195	    p.add_argument("--version", action="version", version=f"%(prog)s {__version__}")
   196	    sub = p.add_subparsers(dest="command", required=True)
   197	    q = sub.add_parser("init", help="Create a portable project configuration.")
   198	    q.add_argument("project_dir")
   199	    q.add_argument("--force", action="store_true")
   200	    q.set_defaults(func=cmd_init)
   201	    q = sub.add_parser("config-check")
   202	    q.add_argument("--config", required=True)
   203	    q.set_defaults(func=cmd_config_check)
   204	    q = sub.add_parser("plan")
   205	    q.add_argument("--config", required=True)
   206	    q.add_argument("--ndate", type=int, default=None)
   207	    q.set_defaults(func=cmd_plan)
   208	    q = sub.add_parser("doctor")
   209	    q.add_argument("--config", required=True)
   210	    q.set_defaults(func=cmd_doctor)
   211	    q = sub.add_parser("run", help="Run the production InSAR processing pipeline.")
   212	    q.add_argument("--config", required=True)
   213	    q.add_argument("--from-stage", default=None)
   214	    q.add_argument("--to-stage", default=None)
   215	    q.add_argument("--dry-run", action="store_true")
   216	    q.add_argument("--force", action="store_true")
   217	    q.add_argument("--list-stages", action="store_true")
   218	    q.set_defaults(func=cmd_run)
   219	    return p
   220	
   221	
   222	def main(argv=None):
   223	    args = build_parser().parse_args(argv)
   224	    return args.func(args)

   228	    main()

    15	def find_config(path: str | Path | None = None) -> Path:
    16	    if path is not None:
    17	        p = Path(path).expanduser().resolve()
    18	
    19	        if not p.is_file():
    20	            raise FileNotFoundError(
    21	                f"Configuration file not found: {p}"
    22	            )
    23	
    24	        return p
    25	
    26	    for name in CONFIG_NAMES:
    27	        p = (Path.cwd() / name).resolve()
    28	
    29	        if p.is_file():
    30	            return p
    31	
    32	    raise FileNotFoundError(
    33	        "No pypsds.yaml configuration file was found."
    34	    )
    35	
    36	
    37	def load_config(
    38	    path: str | Path | None = None,
    39	) -> tuple[dict[str, Any], Path]:
    40	
    41	    p = find_config(path)
    42	
    43	    raw = yaml.safe_load(
    44	        p.read_text(encoding="utf-8")
    45	    ) or {}
    46	
    47	    if not isinstance(raw, dict):
    48	        raise ValueError(
    49	            "Configuration top level must be a mapping."
    50	        )
    51	
    52	    schema = raw.get(
    53	        "schema_version",
    54	        None,
    55	    )
    56	
    57	    if schema != 1:
    58	        raise ValueError(
    59	            f"Unsupported schema_version={schema!r}; "
    60	            "pyPSDS-GAMMA v1.0 requires schema_version: 1"
    61	        )
    62	
    63	    return raw, p
    64	
    65	
    66	def cfg_get(
    67	    cfg: dict[str, Any],
    68	    dotted: str,
    69	    default: Any = None,
    70	) -> Any:
    71	
    72	    value: Any = cfg
    73	
    74	    for part in dotted.split("."):
    75	
    76	        if (
    77	            not isinstance(value, dict)
    78	            or
    79	            part not in value
    80	        ):
    81	            return default
    82	
    83	        value = value[part]
    84	
    85	    return value

    GeometryInputError,

from pypsds.project import ProjectPaths


DATE = "20200102"


def write_par(path, width, length):
    path.write_text(
        f"range_samples: {width}\n"
        f"azimuth_lines: {length}\n",
        encoding="utf-8",
    )


def write_float_raster(path, width, length):
    path.write_bytes(
        b"\x00" * (width * length * 4)
    )


def make_project(tmp_path):
    work = tmp_path / "project"
    data = work / "data"
    rslc = data / "RSLC"
    dem = data / "DEM_prep"
    output = work / "output"

    rslc.mkdir(parents=True)
    dem.mkdir(parents=True)
    output.mkdir(parents=True)

    tab = data / "RSLC_tab"
    tab.write_text("", encoding="utf-8")

    paths = ProjectPaths(
        work_dir=work,
        data_dir=data,
        rslc_dir=rslc,
        rslc_tab=tab,
        output_dir=output,
        dem_dir=dem,
    )

    return paths, rslc, dem
